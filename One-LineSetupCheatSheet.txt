# Clone repo
git clone https://github.com/MarvinOS-online/fitlab-core.git
cd fitlab-core

# Build & start stack (one-shot config containers run automatically)
docker compose up --build -d

# Access UI
# LLM & RAG Chat + Image UI: http://localhost:8080
# HTTPS Ingress: https://localhost
# Stable Diffusion API: http://localhost:7860

# Restart / Stop stack
# docker compose restart
# docker compose down

# Pull additional Ollama models (inside container)
# Example:
# docker compose exec ollama ollama pull llama3.2:3b
# docker compose exec ollama ollama pull mistral:7b
# docker compose exec ollama ollama pull nomic-embed-text:v1.5

# List installed Ollama models
# docker compose exec ollama ollama list

# View logs (follow real-time)
# docker compose logs -f open-webui
# docker compose logs -f ollama
# docker compose logs -f qdrant
